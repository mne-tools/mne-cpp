## Build and run

Needs the [MNE sample dataset](../../manual/sample-dataset.mdx) in `resources/data/MNE-sample-data`.

```bash
cmake --build build/developer-dynamic --target ex_read_evoked
./out/Release/examples/ex_read_evoked --help
```

### Options

| Option | Description |

namespace
{

/** Writes a butterfly plot of the channels of @p evoked of @p kind measured in @p unit, scaled by @p scale. */
void saveButterfly(const FiffEvoked& evoked, int kind, int unit, double scale, const QString& name, const QString& yLabel, const QColor& color)
{
    QVector<double> times(evoked.times.size());
    for (int s = 0; s < times.size(); ++s) {
        times[s] = 1000.0 * evoked.times(s);
    }
    LinePlot plot;
    plot.setTitle(QString("%1 (%2 epochs)").arg(evoked.comment).arg(evoked.nave));
    plot.setXLabel("Time [ms]");
    plot.setYLabel(yLabel);
    for (int c = 0; c < evoked.data.rows(); ++c) {
        if (evoked.info.chs[c].kind != kind || evoked.info.chs[c].unit != unit || evoked.info.bads.contains(evoked.info.ch_names[c])) {
            continue;
        }
        QVector<double> values(times.size());
        for (int s = 0; s < times.size(); ++s) {
            values[s] = scale * evoked.data(c, s);
        }
        plot.addSeries(times, values, QString(), color);
    }
    FigureExporter::save(plot, name, QString("%1 of the '%2' response: one line per channel, baseline-corrected, bad channels excluded.").arg(yLabel.section(' ', 0, 0), evoked.comment));
}

} // namespace

//=============================================================================================================
// MAIN
//=============================================================================================================

//=============================================================================================================

 * The function main marks the entry point of the program.
 * By default, main has the storage class extern.
 *

 * @return the value that was set to exit() (which is 0 if exit() is called via quit()).
 */
int main(int argc, char* argv[])
{
    qInstallMessageHandler(MNELogger::customLogWriter);
    QApplication a(argc, argv);

    // Command Line Parser
    QCommandLineParser parser;
    parser.setApplicationDescription("Read Evoked Example");
    parser.addHelpOption();

    QCommandLineOption evokedFileOption("ave", "Path to the evoked/average <file>.", "file", QCoreApplication::applicationDirPath() + "/../resources/data/MNE-sample-data/MEG/sample/sample_audvis-ave.fif");
    QCommandLineOption evokedIdxOption("aveIdx", "The average <index> to choose from the average file.", "index", "2");
    QCommandLineOption useCTFCompOption("useCTFComp", "Use the CTF compensator, if available.");

    parser.addOption(evokedFileOption);
    parser.addOption(evokedIdxOption);

    parser.process(a);

    //generate FiffEvoked object
    QFile t_sampleFile(parser.value(evokedFileOption));
    FiffEvoked p_FiffEvoked(t_sampleFile, QVariant(parser.value(evokedIdxOption)));

    //Select the head coordinate system
    bool use_ctf_head = parser.isSet(useCTFCompOption);
    FiffCoordTrans meg_trans;

    if (use_ctf_head) {
        if (p_FiffEvoked.info.ctf_head_t.isEmpty())
            std::cout << "\nNo CTF head transformation available" << std::endl;
        else {
            meg_trans = p_FiffEvoked.info.dev_ctf_t;
            FiffCoordTrans eeg_trans(meg_trans);
            eeg_trans.invert_transform();
            std::cout << "Employing the CTF/4D head coordinate system\n"
                      << std::endl;
        }
    } else {
        meg_trans = p_FiffEvoked.info.dev_head_t;
        FiffCoordTrans eeg_trans;
        std::cout << "Employing the Neuromag head coordinate system\n"
                  << std::endl;
    }

    //Transform coil and electrode locations to the desired coordinate frame

    //Create the coil definitions
    //ToDo: MATLAB root fct mne_add_coil_defs needs to be implemented

    //N.B. If a nonstandard (in MNE sense) coil def file is used, do

    std::cout << "Condition '" << p_FiffEvoked.comment.toStdString() << "': " << p_FiffEvoked.nave << " epochs, "
              << p_FiffEvoked.info.nchan << " channels, " << p_FiffEvoked.times.size() << " samples from "
              << p_FiffEvoked.times(0) << " s to " << p_FiffEvoked.times(p_FiffEvoked.times.size() - 1) << " s" << std::endl;

    // Plot a copy corrected over the prestimulus interval, as mne.Evoked.plot shows it

    QPair<float, float> baseline(p_FiffEvoked.times(0), 0.0f);
    corrected.applyBaselineCorrection(baseline);
    saveButterfly(corrected, FIFFV_EEG_CH, FIFF_UNIT_V, 1e6, "eeg", "EEG [µV]", QColor(150, 40, 40, 90));
    saveButterfly(corrected, FIFFV_MEG_CH, FIFF_UNIT_T_M, 1e13, "gradiometers", "Gradiometers [fT/cm]", QColor(30, 60, 110, 70));
    saveButterfly(corrected, FIFFV_MEG_CH, FIFF_UNIT_T, 1e15, "magnetometers", "Magnetometers [fT]", QColor(30, 110, 70, 90));

    return 0;
}
